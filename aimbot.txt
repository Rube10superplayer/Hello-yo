local function LoadConfig()
    if not readfile or not isfile or not isfile(ConfigFileName) then return false end
    return pcall(function()
        local data = HttpService:JSONDecode(readfile(ConfigFileName))
        for key, value in pairs(data) do

                Config[key] = value
            end
        end
    end)

local HookActive = false

local function InitializeHooks()
    if HookActive then return true end
    
    return pcall(function()
        local OldNamecall
        OldNamecall = hookmetamethod(game, "__namecall", newcclosure(function(Self, ...)
            local Method = getnamecallmethod()
            local Args = {...}
            
            if not checkcaller() then
                if Method == "FireServer" or Method == "InvokeServer" then
                    local RemoteName = tostring(Self)
                    
                    for _, blocked in ipairs(Evasion.BlockedRemotes) do
                        if RemoteName:lower():find(blocked:lower()) then
                            return wait(9e9)
                        end
                    end
                    
                    if RemoteName:find("Combat") or RemoteName:find("Damage") or RemoteName:find("Hit") then
                        for i, arg in ipairs(Args) do
                            if typeof(arg) == "Vector3" then
                                local noise = Vector3.new(
                                    math.random(-25, 25) / 1000,
                                    math.random(-25, 25) / 1000,
                                    math.random(-25, 25) / 1000
                                )
                                Args[i] = arg + noise
                            end
                        end
                    end
                end
            end
            
            return OldNamecall(Self, unpack(Args))
        end))
        
        local OldIndex
        OldIndex = hookmetamethod(game, "__index", newcclosure(function(Self, Key)
            if not checkcaller() then
                if Self:IsA("Humanoid") then
                    if Key == "WalkSpeed" then return 16 end
                    if Key == "JumpPower" then return 50 end
                end
                
                if Self == Camera and (Key == "CFrame" or Key == "Focus") then
                    local real = OldIndex(Self, Key)
                    if typeof(real) == "CFrame" then
                        local jitter = CFrame.Angles(
                            math.rad(math.random(-1, 1) / 150),
                            math.rad(math.random(-1, 1) / 150),
                            0
                        )
                        return real * jitter
                    end
                    return real
                end
            end
            
            return OldIndex(Self, Key)
        end))
        
        HookActive = true
    end)
end

local function RemoveESP(player)
    if not ESPObjects[player] then return end
    
    pcall(function()
        for _, obj in pairs(ESPObjects[player]) do
            if typeof(obj) ~= "table" and obj.Remove then
                obj:Remove()
            end
        end
        ESPObjects[player] = nil
    end)
end

local function UpdateESP()
    if not Config.ESPEnabled or not HasDrawing then return end
    
    for player, esp in pairs(ESPObjects) do
        pcall(function()
            if not player or not player.Parent or not player.Character then
                esp.Box.Visible = false
                esp.Tracer.Visible = false
                esp.HealthBar.Visible = false
                esp.HealthBarBG.Visible = false
                esp.Name.Visible = false
                esp.Distance.Visible = false
                return
            end
            
            local char = player.Character
            local root = char:FindFirstChild("HumanoidRootPart")
            local humanoid = char:FindFirstChildOfClass("Humanoid")
            
            if not root or not humanoid or humanoid.Health <= 0 then
                esp.Box.Visible = false
                esp.Tracer.Visible = false
                esp.HealthBar.Visible = false
                esp.HealthBarBG.Visible = false
                esp.Name.Visible = false
                esp.Distance.Visible = false
                return
            end
            
            local distance = (root.Position - Camera.CFrame.Position).Magnitude
            if distance > Config.ESPMaxDistance then
                esp.Box.Visible = false
                esp.Tracer.Visible = false
                esp.HealthBar.Visible = false
                esp.HealthBarBG.Visible = false
                esp.Name.Visible = false
                esp.Distance.Visible = false
                return
            end
            
            local screenPos, onScreen = Camera:WorldToViewportPoint(root.Position)
            
            if not onScreen then
                esp.Box.Visible = false
                esp.Tracer.Visible = false
                esp.HealthBar.Visible = false
                esp.HealthBarBG.Visible = false
                esp.Name.Visible = false
                esp.Distance.Visible = false
                return
            end
            
            local boxWidth = 4000 / distance
            local boxHeight = 6000 / distance
            
            local teamColor = IsValidTarget(player) and Color3.fromRGB(255, 60, 60) or Color3.fromRGB(60, 255, 60)
            
            if Config.ESPBoxes then
                esp.Box.Size = Vector2.new(boxWidth, boxHeight)
                esp.Box.Position = Vector2.new(screenPos.X - boxWidth / 2, screenPos.Y - boxHeight / 2)
                esp.Box.Color = teamColor
                esp.Box.Visible = true
            else
                esp.Box.Visible = false
            end
            
            if Config.ESPTracers then
                local screenCenter = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y)
                esp.Tracer.From = screenCenter
                esp.Tracer.To = Vector2.new(screenPos.X, screenPos.Y)
                esp.Tracer.Color = teamColor
                esp.Tracer.Visible = true
            else
                esp.Tracer.Visible = false
            end
            
            if Config.ESPHealth then
                local healthPercent = humanoid.Health / humanoid.MaxHealth
                local barHeight = boxHeight
                local barWidth = 4
                
                esp.HealthBarBG.Size = Vector2.new(barWidth, barHeight)
                esp.HealthBarBG.Position = Vector2.new(screenPos.X - boxWidth / 2 - 6, screenPos.Y - boxHeight / 2)
                esp.HealthBarBG.Visible = true
                
                esp.HealthBar.Size = Vector2.new(barWidth, barHeight * healthPercent)
                esp.HealthBar.Position = Vector2.new(screenPos.X - boxWidth / 2 - 6, screenPos.Y - boxHeight / 2 + barHeight * (1 - healthPercent))
                
                if healthPercent > 0.6 then
                    esp.HealthBar.Color = Color3.fromRGB(0, 255, 0)
                elseif healthPercent > 0.3 then
                    esp.HealthBar.Color = Color3.fromRGB(255, 255, 0)
                else
                    esp.HealthBar.Color = Color3.fromRGB(255, 0, 0)
                end
                
                esp.HealthBar.Visible = true
            else
                esp.HealthBar.Visible = false
                esp.HealthBarBG.Visible = false
            end
            
            if Config.ESPNames then
                esp.Name.Text = player.Name
                esp.Name.Position = Vector2.new(screenPos.X, screenPos.Y - boxHeight / 2 - 18)
                esp.Name.Color = teamColor
                esp.Name.Visible = true
            else
                esp.Name.Visible = false
            end
            
            if Config.ESPDistance then
                esp.Distance.Text = math.floor(distance) .. "m"
                esp.Distance.Position = Vector2.new(screenPos.X, screenPos.Y + boxHeight / 2 + 5)
                esp.Distance.Visible = true
            else
                esp.Distance.Visible = false
            end
        end)
    end
end

-- Targeting System
local VelocityHistory = {}

local function HasPvPEnabled(player)
    if not Config.PvPCheck then return true end
    
    return pcall(function()
        local pvpValue = player:FindFirstChild("PvP") or player:FindFirstChild("IsPvP")
        if pvpValue and pvpValue:IsA("BoolValue") then
            return pvpValue.Value
        end
        
        local char = player.Character
        if char then
            local tags = {"Combat", "PvPEnabled", "InCombat", "PvPMode"}
            for _, tagName in ipairs(tags) do
                local tag = char:FindFirstChild(tagName)
                if tag then
                    return tag:IsA("BoolValue") and tag.Value or true
                end
            end
            
            local head = char:FindFirstChild("Head")
            if head then
                for _, obj in ipairs(head:GetChildren()) do
                    if obj:IsA("BillboardGui") then
                        local label = obj:FindFirstChildOfClass("TextLabel")
                        if label then
                            local color = label.TextColor3
                            if color.R > 0.7 and color.G < 0.4 then
                                return true
                            end
                        end
                    end
                end
            end
        end
        
        return false
    end) and true or false
end

local function GetTeam(player)
    local success, result = pcall(function()
        if player.Team then return player.Team.Name end
        
        local char = player.Character
        if char then
            if char:FindFirstChild("IsMarineTag") or char:FindFirstChild("Marine") then
                return "Marines"
            end
            if char:FindFirstChild("IsPirateTag") or char:FindFirstChild("Pirate") then
                return "Pirates"
            end
        end
        
        return "Unallied"
    end)
    
    return success and result or "Unallied"
end

function IsValidTarget(player)
    if not player or player == LocalPlayer then return false end
    
    local success, result = pcall(function()
        if not HasPvPEnabled(player) then return false end
        if not Config.TeamCheck then return true end
        
        local localTeam = GetTeam(LocalPlayer)
        local targetTeam = GetTeam(player)
        
        if localTeam == "Pirates" then
            return targetTeam == "Marines" or targetTeam == "Unallied"
        elseif localTeam == "Marines" then
            return targetTeam == "Pirates"
        end
        
        return true
    end)
    
    return success and result or false
end

local function IsVisible(targetPart)
    if not Config.VisibleCheck then return true end
    
    return pcall(function()
        local char = LocalPlayer.Character
        if not char then return false end
        
        local origin = Camera.CFrame.Position
        local direction = targetPart.Position - origin
        
        local rayParams = RaycastParams.new()
        rayParams.FilterType = Enum.RaycastFilterType.Blacklist
        rayParams.FilterDescendantsInstances = {char, targetPart.Parent}
        rayParams.IgnoreWater = true
        
        local result = workspace:Raycast(origin, direction, rayParams)
        
        if not result then return true end
        if result.Instance.Transparency > 0.8 then return true end
        
        return result.Instance:IsDescendantOf(targetPart.Parent)
    end) and true or false
end

local function UpdateVelocityHistory(player, velocity, position)
    if not VelocityHistory[player.UserId] then
        VelocityHistory[player.UserId] = {}
    end
    
    local history = VelocityHistory[player.UserId]
    table.insert(history, {velocity = velocity, position = position, time = tick()})
    
    while #history > 20 do
        table.remove(history, 1)
    end
end

local function GetResolvedVelocity(player)
    return pcall(function()
        local char = player.Character
        if not char then return Vector3.zero end
        
        local root = char:FindFirstChild("HumanoidRootPart")
        if not root then return Vector3.zero end
        
        local currentVel = root.AssemblyVelocity
        UpdateVelocityHistory(player, currentVel, root.Position)
        
        if not Config.VelocityResolver then return currentVel end
        
        local history = VelocityHistory[player.UserId]
        if not history or #history < 5 then return currentVel end
        
        local recent = history[#history]
        local old = history[math.max(1, #history - 4)]
        local timeDelta = recent.time - old.time
        
        if timeDelta == 0 then return currentVel end
        
        local acceleration = (recent.velocity - old.velocity) / timeDelta
        
        if Config.StrafePrediction then
            local strafeChanges = 0
            for i = math.max(1, #history - 3), #history do
                local curr = history[i].velocity
                local prev = history[i - 1] and history[i - 1].velocity or curr
                
                if (curr.X * prev.X < 0) or (curr.Z * prev.Z < 0) then
                    strafeChanges = strafeChanges + 1
                end
            end
            
            if strafeChanges >= 2 then
                acceleration = Vector3.new(
                    acceleration.X * 0.25,
                    acceleration.Y,
                    acceleration.Z * 0.25
                )
            end
        end
        
        return currentVel + (acceleration * Config.PredictionStrength * 0.8)
    end) and Vector3.zero or Vector3.zero
end

local function PredictJumpArc(player, targetPart)
    if not Config.JumpPrediction then return Vector3.zero end
    
    return pcall(function()
        local humanoid = player.Character:FindFirstChildOfClass("Humanoid")
        if not humanoid then return Vector3.zero end
        
        local state = humanoid:GetState()
        
        if state == Enum.HumanoidStateType.Freefall or 
           state == Enum.HumanoidStateType.Jumping then
            
            local velocity = targetPart.AssemblyVelocity
            local gravity = workspace.Gravity
            
            local timeToApex = math.abs(velocity.Y) / gravity
            local apexHeight = velocity.Y * timeToApex - (0.5 * gravity * timeToApex^2)
            
            return Vector3.new(0, apexHeight * 0.42, 0)
        end
        
        return Vector3.zero
    end) and Vector3.zero or Vector3.zero
end

local function GetDynamicSmoothness(distance, velocity)
    local baseSmoothness = Config.BaseSmoothness
    
    if Config.DistanceSmoothing then
        local distanceFactor = math.clamp(distance / 100, 0.65, 1.6)
        baseSmoothness = baseSmoothness * distanceFactor
    end
    
    if Config.VelocitySmoothing and Config.AdaptiveSmoothing then
        local speed = velocity.Magnitude
        if speed > 35 then
            baseSmoothness = baseSmoothness * 0.82
        elseif speed < 8 then
            baseSmoothness = baseSmoothness * 1.18
        end
    end
    
    return math.clamp(baseSmoothness, 0.05, 0.35)
end

local function GetFOVMultiplier(distanceFromCenter)
    if not Config.FOVFalloff then return 1 end
    
    local normalized = distanceFromCenter / Config.FOVRadius
    
    if normalized > 0.7 then
        local edge = (normalized - 0.7) / 0.3
        return 1 - (edge^2 * 0.65)
    end
    
    return 1
end

local CameraState = {
    position = Vector3.zero,
    velocity = Vector3.zero,
    lastUpdate = tick()
}

local function ApplyShakeReduction(targetPos)
    return pcall(function()
        local dt = tick() - CameraState.lastUpdate
        CameraState.lastUpdate = tick()
        
        if dt > 0.1 then dt = 0.016 end
        
        local predicted = CameraState.position + (CameraState.velocity * dt)
        local innovation = targetPos - predicted
        
        local alpha = Config.ShakeReduction
        CameraState.position = predicted + (innovation * (1 - alpha))
        CameraState.velocity = innovation / dt
        
        return CameraState.position
    end) and CameraState.position or targetPos
end

local function GetBestHitbox(char, distance, velocity)
    if not Config.SmartHitbox then
        return char:FindFirstChild(Config.HitboxPriority[1]) or char:FindFirstChild("HumanoidRootPart")
    end
    
    local speed = velocity.Magnitude
    
    if distance < 35 and speed < 12 then
        return char:FindFirstChild("Head")
    end
    
    if distance > 70 or speed > 28 then
        return char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("UpperTorso")
    end
    
    for _, hitboxName in ipairs(Config.HitboxPriority) do
        local part = char:FindFirstChild(hitboxName)
        if part then return part end
    end
    
    return char:FindFirstChild("HumanoidRootPart")
end

local function GetTargetPriority(player)
    if not Config.PrioritySystem then return 1 end
    
    return pcall(function()
        local char = player.Character
        if not char then return 1 end
        
        local humanoid = char:FindFirstChildOfClass("Humanoid")
        if not humanoid then return 1 end
        
        local healthPercent = humanoid.Health / humanoid.MaxHealth
        
        if healthPercent < 0.35 then return 2.2 end
        if healthPercent < 0.6 then return 1.5 end
        
        return 1.0
    end) and 1 or 1
end

local function ApplyHumanization(targetPos)
    return pcall(function()
        local t = tick()
        local microshake = Vector3.new(
            (math.sin(t * 47) * math.cos(t * 23) - 0.5) * BehaviorProfile.Microshake,
            (math.cos(t * 31) * math.sin(t * 41) - 0.5) * BehaviorProfile.Microshake,
            (math.sin(t * 37) * math.sin(t * 29) - 0.5) * BehaviorProfile.Microshake
        )
        
        local overshoot = Vector3.zero
        if math.random(1, 100) <= BehaviorProfile.OvershootChance then
            local dir = (targetPos - Camera.CFrame.Position).Unit
            overshoot = dir * (math.random(8, 18) / 100)
        end
        
        return targetPos + microshake + overshoot
    end) and targetPos or targetPos
end

-- Evasion Functions
local function ShouldMiss()
    if math.random(1, 100) <= Evasion.MissChance then return true end
    if Evasion.CurrentLockDuration > Evasion.MaxLockTime then return true end
    
    local headshotRatio = Evasion.Headshots / math.max(Evasion.TotalLocks, 1)
    if headshotRatio > 0.6 then
        return math.random(1, 100) <= 25
    end
    
    return false
end

local lastCameraAngles = Vector2.zero
local lastAngleTime = tick()

local function LimitAngularVelocity(targetCFrame)
    local currentAngles = Vector2.new(
        math.deg(math.asin(targetCFrame.LookVector.Y)),
        math.deg(math.atan2(targetCFrame.LookVector.X, targetCFrame.LookVector.Z))
    )
    
    local deltaTime = tick() - lastAngleTime
    if deltaTime == 0 then return targetCFrame end
    
    local angleDelta = (currentAngles - lastCameraAngles)
    local angularVelocity = angleDelta.Magnitude / deltaTime
    
    if angularVelocity > Evasion.MaxAngularVelocity then
        local limitFactor = Evasion.MaxAngularVelocity / angularVelocity
        local limitedDelta = angleDelta * limitFactor
        local newAngles = lastCameraAngles + limitedDelta
        
        local yaw = math.rad(newAngles.Y)
        local pitch = math.rad(newAngles.X)
        
        targetCFrame = CFrame.new(Camera.CFrame.Position) * 
                      CFrame.Angles(0, yaw, 0) * 
                      CFrame.Angles(pitch, 0, 0)
    end
    
    lastCameraAngles = currentAngles
    lastAngleTime = tick()
    
    return targetCFrame
end

local lockStartTime = 0

local function UpdateLockDuration(hasTarget)
    if hasTarget then
        if lockStartTime == 0 then
            lockStartTime = tick()
        end
        Evasion.CurrentLockDuration = tick() - lockStartTime
    else
        lockStartTime = 0
        Evasion.CurrentLockDuration = 0
    end
end

local function CheckSuspicionLevel()
    if Evasion.SuspiciousActions >= Evasion.FlagThreshold then
        Config.Enabled = false
        
        pcall(function()
            game:GetService("StarterGui"):SetCore("SendNotification", {
                Title = "⚠️ AUTO-DISABLED",
                Text = "Cooling down for 30s",
                Duration = 5
            })
        end)
        
        task.delay(30, function()
            Evasion.SuspiciousActions = 0
        end)
    end
end

local function FlagSuspicious(reason)
    Evasion.SuspiciousActions = Evasion.SuspiciousActions + 1
    Evasion.LastFlagTime = tick()
    CheckSuspicionLevel()
end

-- Target Acquisition
local currentTarget = nil
local targetLockTime = 0
local lastTargetSwitch = 0

local function GetNearestTarget()
    return pcall(function()
        if tick() - Evasion.LastTargetSwitch < Evasion.MinSwitchDelay then
            return currentTarget
        end
        
        if ShouldMiss() and math.random(1, 100) <= 50 then
            Evasion.LastTargetSwitch = tick()
            return nil
        end
        
        if Evasion.CurrentLockDuration > Evasion.MaxLockTime then
            FlagSuspicious("Lock duration")
            Evasion.LastTargetSwitch = tick()
            return nil
        end
        
        local nearestPlayer = nil
        local shortestDistance = math.huge
        local screenCenter = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
        
        if Config.StickyTarget and currentTarget then
            local timeLocked = tick() - targetLockTime
            
            if timeLocked < Config.StickyThreshold and IsValidTarget(currentTarget) then
                local char = currentTarget.Character
                if char then
                    local root = char:FindFirstChild("HumanoidRootPart")
                    if root then
                        local humanoid = char:FindFirstChildOfClass("Humanoid")
                        if humanoid and humanoid.Health > 0 then
                            local screenPos, onScreen = Camera:WorldToViewportPoint(root.Position)
                            if onScreen then
                                local dist2D = (Vector2.new(screenPos.X, screenPos.Y) - screenCenter).Magnitude
                                if dist2D < Config.FOVRadius * 1.35 then
                                    return currentTarget
                                end
                            end
                        end
                    end
                end
            end
        end
        
        for _, player in pairs(Players:GetPlayers()) do
            if IsValidTarget(player) then
                local char = player.Character
                if char then
                    local root = char:FindFirstChild("HumanoidRootPart")
                    if root then
                        local distance3D = (root.Position - Camera.CFrame.Position).Magnitude
                        local velocity = root.AssemblyVelocity
                        local hitbox = GetBestHitbox(char, distance3D, velocity)
                        
                        if hitbox then
                            local humanoid = char:FindFirstChildOfClass("Humanoid")
                            
                            if humanoid and humanoid.Health > 0 then
                                local screenPos, onScreen = Camera:WorldToViewportPoint(hitbox.Position)
                                
                                if onScreen then
                                    local screenVec = Vector2.new(screenPos.X, screenPos.Y)
                                    local distance2D = (screenVec - screenCenter).Magnitude
                                    
                                    local priority = GetTargetPriority(player)
                                    local weightedDist = distance2D / priority
                                    
                                    if distance2D < Config.FOVRadius and weightedDist < shortestDistance then
                                        if IsVisible(hitbox) then
                                            nearestPlayer = player
                                            shortestDistance = weightedDist
                                        end
                                    end
                                end
                            end
                        end
                    end
                end
            end
        end
        
        if nearestPlayer and nearestPlayer ~= currentTarget then
            targetLockTime = tick()
            Evasion.LastTargetSwitch = tick()
            Evasion.TotalLocks = Evasion.TotalLocks + 1
        end
        
        return nearestPlayer
    end) and currentTarget or nil
end

-- Main Camlock Loop
local lastUpdateTime = 0
local performanceStats = {
    avgFrameTime = 0,
    updateCount = 0
}

RunService.RenderStepped:Connect(function(deltaTime)
    pcall(function()
        if not Config.Enabled then 
            UpdateLockDuration(false)
            return 
        end
        
        local targetInterval = 1 / 60
        local variance = BehaviorProfile.UpdateVariance
        local randomInterval = targetInterval + (math.random() * variance * 2 - variance)
        
        if tick() - lastUpdateTime < randomInterval then return end
        lastUpdateTime = tick()
        
        if tick() - BehaviorProfile.LastUpdate > math.random(35, 65) then
            GenerateBehaviorProfile()
        end
        
        local reactionDelay = Evasion.NaturalDelay + (math.random(-20, 20) / 1000)
        task.wait(reactionDelay)
        
        currentTarget = GetNearestTarget()
        UpdateLockDuration(currentTarget ~= nil)
        
        if currentTarget then
            local char = currentTarget.Character
            if char then
                local root = char:FindFirstChild("HumanoidRootPart")
                if root then
                    local distance3D = (root.Position - Camera.CFrame.Position).Magnitude
                    local velocity = GetResolvedVelocity(currentTarget)
                    local hitbox = GetBestHitbox(char, distance3D, velocity)
                    
                    if hitbox then
                        local humanoid = char:FindFirstChildOfClass("Humanoid")
                        
                        if humanoid and humanoid.Health > 0 then
                            if hitbox.Name == "Head" then
                                Evasion.Headshots = Evasion.Headshots + 1
                            else
                                Evasion.BodyShots = Evasion.BodyShots + 1
                            end
                            
                            local predictedPos = hitbox.Position + (velocity * Config.PredictionStrength)
                            predictedPos = predictedPos + PredictJumpArc(currentTarget, hitbox)
                            predictedPos = ApplyHumanization(predictedPos)
                            predictedPos = ApplyShakeReduction(predictedPos)
                            
                            local smoothness = GetDynamicSmoothness(distance3D, velocity)
                            local screenPos = Camera:WorldToViewportPoint(predictedPos)
                            local screenCenter = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
                            local distFromCenter = (Vector2.new(screenPos.X, screenPos.Y) - screenCenter).Magnitude
                            local fovMult = GetFOVMultiplier(distFromCenter)
                            
                            local finalSmooth = smoothness * fovMult
                            local targetCFrame = CFrame.new(Camera.CFrame.Position, predictedPos)
                            
                            targetCFrame = LimitAngularVelocity(targetCFrame)
                            
                            Camera.CFrame = Camera.CFrame:Lerp(targetCFrame, finalSmooth)
                            
                            if HasDrawing and TargetIndicator and screenPos then
                                TargetIndicator.Position = Vector2.new(screenPos.X, screenPos.Y)
                                TargetIndicator.Visible = true
                            end
                            
                            performanceStats.updateCount = performanceStats.updateCount + 1
                            performanceStats.avgFrameTime = (performanceStats.avgFrameTime * 0.95) + (deltaTime * 0.05)
                        end
                    end
                end
            end
        else
            if HasDrawing and TargetIndicator then
                TargetIndicator.Visible = false
            end
            
            if Config.DynamicFOV and tick() - lastTargetSwitch > 3.5 then
                local targetFOV = math.min(Config.FOVRadius * 1.55, 450)
                if HasDrawing then
                    FOVCircle.Radius = FOVCircle.Radius + ((targetFOV - FOVCircle.Radius) * 0.06)
                    FOVFill.Radius = FOVCircle.Radius
                end
            else
                if HasDrawing then
                    FOVCircle.Radius = FOVCircle.Radius + ((Config.FOVRadius - FOVCircle.Radius) * 0.12)
                    FOVFill.Radius = FOVCircle.Radius
                end
            end
        end
    end)
end)

-- Mobile Touch Controls
if IsMobile and Config.TouchActivation then
    local lastTap = 0
    local tapCount = 0
    
    UserInputService.TouchTap:Connect(function(touchPositions, gameProcessed)
        if gameProcessed then return end
        
        local pos = touchPositions[1]
        local screenSize = Camera.ViewportSize
        
        if pos.X < Config.ScreenEdgeDeadzone or 
           pos.X > screenSize.X - Config.ScreenEdgeDeadzone or
           pos.Y < Config.ScreenEdgeDeadzone or
           pos.Y > screenSize.Y - Config.ScreenEdgeDeadzone then
            return
        end
        
        if tick() - lastTap < 0.35 then
            tapCount = tapCount + 1
            if tapCount >= 2 then
                Config.Enabled = not Config.Enabled
                
                StatusLabel.Text = Config.Enabled and "● ACTIVE" or "● OFFLINE"
                TweenService:Create(StatusLabel, TweenInfo.new(0.3), {TextColor3 = Config.Enabled and Color3.fromRGB(100, 255, 100) or Color3.fromRGB(255, 70, 70)}):Play()
                TweenService:Create(BorderStroke, TweenInfo.new(0.3), {Color = Config.Enabled and Color3.fromRGB(100, 255, 100) or Color3.fromRGB(255, 70, 70)}):Play()
                
                if HasDrawing then
                    FOVCircle.Color = Config.Enabled and Color3.fromRGB(100, 255, 100) or Color3.fromRGB(255, 255, 255)
                end
                
                tapCount = 0
            end
        else
            tapCount = 1
        end
        
        lastTap = tick()
    end)
end

-- Performance Monitor
RunService.Heartbeat:Connect(function()
    pcall(function()
        if Config.Enabled and currentTarget then
            local char = currentTarget.Character
            if char then
                local humanoid = char:FindFirstChildOfClass("Humanoid")
                if humanoid then
                    local hp = math.floor((humanoid.Health / humanoid.MaxHealth) * 100)
                    TargetLabel.Text = string.format("Target: %s [%d%%]", currentTarget.Name, hp)
                    TweenService:Create(TargetLabel, TweenInfo.new(0.3), {TextColor3 = Color3.fromRGB(100, 255, 100)}):Play()

        else
            TargetLabel.Text = "Target: None"
            TweenService:Create(TargetLabel, TweenInfo.new(0.3), {TextColor3 = Color3.fromRGB(150, 150, 150)}):Play()
        end
        
        local fps = math.floor(1 / math.max(performanceStats.avgFrameTime, 0.001))
        local headshotRatio = Evasion.Headshots / math.max(Evasion.TotalLocks, 1)
        StatsLabel.Text = string.format("FPS: %d | Locks: %d | HS: %.0f%%", fps, Evasion.TotalLocks, headshotRatio * 100)
    end)
